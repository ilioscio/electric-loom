/* =====================================================================
   SHADERS
   ---------------------------------------------------------------------
   THE LOOP RULE, obeyed by every generator below:
   time enters only as TA = 2*PI*T with T in [0,1), and every coefficient
   multiplying TA is an integer (floor()ed in the shader, integer-stepped
   in the UI). Anything periodic in 2*PI is therefore bit-identical at
   T=0 and T=1. Scrolling fields are additionally locked to a whole
   number of tiles per loop. No cross-fades, no cheats.
   ===================================================================== */

const GLSL_PRELUDE = `#version 300 es
precision highp float;
precision highp int;
out vec4 O;
uniform vec2  uRes;
uniform float uT;
uniform float uSeed;
uniform sampler2D uPal;
uniform float uP[16];
uniform float uZoom, uRot, uMirror, uMirrorRot, uPalRot;
uniform vec2  uPan;
uniform float uAlphaMode, uAlphaGain, uAlphaBias;
const float PI  = 3.141592653589793;
const float TAU = 6.283185307179586;
#define T uT
#define TA (TAU*uT)

vec3 pal(float x){ return texture(uPal, vec2(x + uPalRot, 0.5)).rgb; }

float h11(float p){ p=fract(p*0.1031); p*=p+33.33; p*=p+p; return fract(p); }
float h12(vec2 p){ vec3 p3=fract(vec3(p.xyx)*0.1031); p3+=dot(p3,p3.yzx+33.33); return fract((p3.x+p3.y)*p3.z); }
vec2  h22(vec2 p){ vec3 p3=fract(vec3(p.xyx)*vec3(0.1031,0.1030,0.0973)); p3+=dot(p3,p3.yzx+33.33); return fract((p3.xx+p3.yz)*p3.zy); }
float h13(vec3 p3){ p3=fract(p3*0.1031); p3+=dot(p3,p3.zyx+31.32); return fract((p3.x+p3.y)*p3.z); }
float h14(vec4 p4){ p4=fract(p4*vec4(0.1031,0.1030,0.0973,0.1099)); p4+=dot(p4,p4.wzxy+33.33); return fract((p4.x+p4.y)*(p4.z+p4.w)); }

float vn2(vec2 p){
  vec2 i=floor(p), f=fract(p); f=f*f*(3.0-2.0*f);
  return mix(mix(h12(i),h12(i+vec2(1,0)),f.x), mix(h12(i+vec2(0,1)),h12(i+vec2(1,1)),f.x), f.y);
}
float vn3(vec3 p){
  vec3 i=floor(p), f=fract(p); f=f*f*(3.0-2.0*f);
  float a=mix(mix(h13(i),h13(i+vec3(1,0,0)),f.x), mix(h13(i+vec3(0,1,0)),h13(i+vec3(1,1,0)),f.x), f.y);
  float b=mix(mix(h13(i+vec3(0,0,1)),h13(i+vec3(1,0,1)),f.x), mix(h13(i+vec3(0,1,1)),h13(i+vec3(1,1,1)),f.x), f.y);
  return mix(a,b,f.z);
}
float vn4(vec4 p){
  vec4 i=floor(p), f=fract(p); f=f*f*(3.0-2.0*f);
  float r=0.0;
  for(int a=0;a<2;a++)for(int b=0;b<2;b++)for(int c=0;c<2;c++)for(int d=0;d<2;d++){
    vec4 o=vec4(float(a),float(b),float(c),float(d));
    float w=mix(1.0-f.x,f.x,o.x)*mix(1.0-f.y,f.y,o.y)*mix(1.0-f.z,f.z,o.z)*mix(1.0-f.w,f.w,o.w);
    r+=w*h14(i+o);
  }
  return r;
}
float fbm2(vec2 p,int oct){
  float a=0.5,s=0.0,n=0.0;
  for(int i=0;i<8;i++){ if(i>=oct) break; s+=a*vn2(p); n+=a; p=p*2.03+vec2(1.7,9.2); a*=0.5; }
  return s/max(n,1e-5);
}
float fbm4(vec4 p,int oct){
  float a=0.5,s=0.0,n=0.0;
  for(int i=0;i<5;i++){ if(i>=oct) break; s+=a*vn4(p); n+=a; p=p*2.03+vec4(1.7,9.2,4.1,7.3); a*=0.5; }
  return s/max(n,1e-5);
}
vec2 rot(vec2 p,float a){ float c=cos(a),s=sin(a); return vec2(c*p.x-s*p.y, s*p.x+c*p.y); }

/* Integer modulo that survives the drivers.
   GLSL mod(x,y) is x - y*floor(x/y), and several drivers evaluate x/y as
   x*(1/y). With y=6 the reciprocal rounds down, floor(6*0.16666666) is 0,
   and mod(6.0,6.0) hands back 6.0 instead of 0. Any tile index that scrolls
   by a whole period per loop lands on exactly that case at the seam, so it
   has to be computed without trusting the division. Both arguments here are
   integer valued and at least 1/16 apart from a false floor, so the nudge is
   safe by a wide margin. */
vec2 imod(vec2 v,float per){ return v - per*floor(v/per + 1e-3); }

vec4 hexCell(vec2 p){
  vec2 s=vec2(1.0,1.7320508), h=s*0.5;
  vec2 a=mod(p,s)-h, b=mod(p-h,s)-h;
  vec2 gv = dot(a,a)<dot(b,b) ? a : b;
  return vec4(gv, p-gv);
}
float hexDist(vec2 p){ p=abs(p); return max(dot(p,normalize(vec2(1.0,1.7320508))), p.x); }

vec2 kal(vec2 p,float n){
  if(n<1.5) return p;
  float r=length(p);
  float a=atan(p.y,p.x)-uMirrorRot;
  float seg=TAU/n;
  a=mod(a,seg);
  a=abs(a-seg*0.5);
  a+=uMirrorRot;
  return vec2(cos(a),sin(a))*r;
}
vec2 uvBase(){
  vec2 uv=(gl_FragCoord.xy-0.5*uRes)/min(uRes.x,uRes.y);
  uv=kal(uv,uMirror);
  uv=rot(uv,uRot);
  return uv*uZoom+uPan;
}
`;

const GLSL_MAIN = `
void main(){
  vec3 c = field(uvBase());
  c = max(c, 0.0);
  float a = 1.0;
  if(uAlphaMode > 0.5){
    float l = dot(c, vec3(0.2126,0.7152,0.0722));
    a = clamp((l - uAlphaBias) * uAlphaGain, 0.0, 1.0);
  }
  O = vec4(c, a);
}
`;
