/* A native <input type=range> inside a scrolling column is a touch trap:
   touching the track to begin a scroll commits a value change before the
   browser has decided the gesture is a vertical pan, so the page cannot be
   scrolled past a slider without dragging it. touch-action:pan-y is meant to
   fix this but is not honoured on range inputs on every engine (Firefox for
   Android in particular). This resolves the gesture ourselves: a touch is
   ignored until it moves further horizontally than vertically, at which point
   it is a genuine drag; a vertical-dominant move is a scroll and every value
   change it produces is reverted. The input event is caught in the capture
   phase, so the control's own handlers never see a blocked change and nothing
   re-renders while scrolling. Mouse and pen are untouched. */
function guardSliderTouch(inp) {
  let active = false, mode = null, sx = 0, sy = 0, startVal = '';
  inp.addEventListener('touchstart', e => {
    const t = e.touches[0];
    active = true; mode = null; startVal = inp.value; sx = t.clientX; sy = t.clientY;
  }, { passive: true });
  inp.addEventListener('touchmove', e => {
    if (mode) return;
    const t = e.touches[0], dx = Math.abs(t.clientX - sx), dy = Math.abs(t.clientY - sy);
    if (dx > dy && dx > 5) mode = 'slide';
    else if (dy >= dx && dy > 5) mode = 'scroll';
  }, { passive: true });
  const end = () => { active = false; mode = null; };
  inp.addEventListener('touchend', end, { passive: true });
  inp.addEventListener('touchcancel', end, { passive: true });
  inp.addEventListener('input', e => {
    if (active && mode !== 'slide') { inp.value = startVal; e.stopImmediatePropagation(); }
  }, true);
}

let toastTimer = 0;

  const t = $('#toast'); t.textContent = msg; t.classList.add('show');

function fmtBytes(n) {
  if (!isFinite(n)) return '--';
  if (n < 1024) return n + ' B';
  if (n < 1048576) return (n / 1024).toFixed(1) + ' KB';
  if (n < 1073741824) return (n / 1048576).toFixed(1) + ' MB';
  return (n / 1073741824).toFixed(2) + ' GB';
}
function fmtTime(s) {
  if (!isFinite(s) || s < 0) return '--';
  s = Math.round(s);
  const m = Math.floor(s / 60);
  return m > 0 ? m + 'm ' + String(s % 60).padStart(2, '0') + 's' : s + 's';
}

/* ------------------------------------------------------------ seeded rng */
function mulberry32(a) {
  return function () {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}
function hashStr(s) {
  let h = 2166136261 >>> 0;
  for (let i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 16777619) >>> 0; }
  return h >>> 0;
}
function randSeedWord() {
  const a = ['aur','vex','lum','zar','nyx','pho','sol','tri','quo','hex','vor','ely','kry','oni','pyr','umb','cel','dra','ith','oma'];
  const b = ['ora','ith','ane','yss','eon','ura','ix','olt','ade','umi','ess','arn','ovo','ymn','ael','usk'];
  const r = Math.random;
  return a[(r() * a.length) | 0] + b[(r() * b.length) | 0] + ((r() * 900 + 100) | 0);
}

/* ---------------------------------------------------------------- colour */
function hex2rgb(h) {
  h = h.replace('#', '');
  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2];
  const n = parseInt(h, 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
}
function rgb2hex(r, g, b) {
  const c = v => clamp(Math.round(v), 0, 255).toString(16).padStart(2, '0');
  return '#' + c(r) + c(g) + c(b);
}
/* sRGB <-> linear */
const s2l = c => { c /= 255; return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); };
const l2s = c => { c = c <= 0.0031308 ? c * 12.92 : 1.055 * Math.pow(Math.max(c, 0), 1 / 2.4) - 0.055; return clamp(c * 255, 0, 255); };

/* sRGB (0-255) -> OKLab */
function rgb2oklab(r, g, b) {
  const R = s2l(r), G = s2l(g), B = s2l(b);
  const l = Math.cbrt(0.4122214708 * R + 0.5363325363 * G + 0.0514459929 * B);
  const m = Math.cbrt(0.2119034982 * R + 0.6806995451 * G + 0.1073969566 * B);
  const s = Math.cbrt(0.0883024619 * R + 0.2817188376 * G + 0.6299787005 * B);
  return [
    0.2104542553 * l + 0.7936177850 * m - 0.0040720468 * s,
    1.9779984951 * l - 2.4285922050 * m + 0.4505937099 * s,
    0.0259040371 * l + 0.7827717662 * m - 0.8086757660 * s
  ];
}
function oklab2rgb(L, a, bb) {
  const l_ = L + 0.3963377774 * a + 0.2158037573 * bb;
  const m_ = L - 0.1055613458 * a - 0.0638541728 * bb;
  const s_ = L - 0.0894841775 * a - 1.2914855480 * bb;
  const l = l_ * l_ * l_, m = m_ * m_ * m_, s = s_ * s_ * s_;
  return [
    l2s(+4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s),
    l2s(-1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s),
    l2s(-0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s)
  ];
}
function rgb2hsl(r, g, b) {
  r /= 255; g /= 255; b /= 255;
  const mx = Math.max(r, g, b), mn = Math.min(r, g, b), d = mx - mn;
  let h = 0; const l = (mx + mn) / 2;
  const s = d === 0 ? 0 : d / (1 - Math.abs(2 * l - 1));
  if (d !== 0) {
    if (mx === r) h = ((g - b) / d) % 6;
    else if (mx === g) h = (b - r) / d + 2;
    else h = (r - g) / d + 4;
    h /= 6; if (h < 0) h += 1;
  }
  return [h, s, l];
}
function hsl2rgb(h, s, l) {
  h = ((h % 1) + 1) % 1;
  const c = (1 - Math.abs(2 * l - 1)) * s, x = c * (1 - Math.abs((h * 6) % 2 - 1)), m = l - c / 2;
  let r = 0, g = 0, b = 0; const seg = Math.floor(h * 6);
  if (seg === 0) { r = c; g = x; } else if (seg === 1) { r = x; g = c; }
  else if (seg === 2) { g = c; b = x; } else if (seg === 3) { g = x; b = c; }
  else if (seg === 4) { r = x; b = c; } else { r = c; b = x; }
  return [(r + m) * 255, (g + m) * 255, (b + m) * 255];
}

/* Build a 512-entry cyclic LUT from stops, blending in the chosen space. */
const LUT_N = 512;
function buildLut(stops, space) {
  const n = stops.length;
  const S = stops.slice().sort((a, b) => a.p - b.p);
  const out = new Uint8Array(LUT_N * 4);
  if (n === 0) return out;
  if (n === 1) {
    const c = hex2rgb(S[0].c);
    for (let i = 0; i < LUT_N; i++) { out[i * 4] = c[0]; out[i * 4 + 1] = c[1]; out[i * 4 + 2] = c[2]; out[i * 4 + 3] = 255; }
    return out;
  }
  const conv = space === 'oklab' ? (c => rgb2oklab(c[0], c[1], c[2]))
             : space === 'hsl'   ? (c => rgb2hsl(c[0], c[1], c[2]))
             : (c => c);
  const back = space === 'oklab' ? (v => oklab2rgb(v[0], v[1], v[2]))
             : space === 'hsl'   ? (v => hsl2rgb(v[0], v[1], v[2]))
             : (v => v);
  const V = S.map(s => conv(hex2rgb(s.c)));
  // HSL needs shortest-arc hue unwrapping so a rainbow does not run backwards
  if (space === 'hsl') {
    for (let i = 1; i < V.length; i++) {
      let d = V[i][0] - V[i - 1][0];
      if (d > 0.5) V[i][0] -= 1; else if (d < -0.5) V[i][0] += 1;
    }
  }
  for (let i = 0; i < LUT_N; i++) {
    const t = i / LUT_N;
    let k = 0;
    while (k < n - 1 && S[k + 1].p <= t) k++;
    let a, b, va, vb, f;
    if (t < S[0].p) {           // wrap segment: last -> first
      a = S[n - 1].p - 1; b = S[0].p; va = V[n - 1]; vb = V[0];
    } else if (k === n - 1) {
      a = S[n - 1].p; b = S[0].p + 1; va = V[n - 1]; vb = V[0];
    } else {
      a = S[k].p; b = S[k + 1].p; va = V[k]; vb = V[k + 1];
    }
    f = b - a < 1e-9 ? 0 : (t - a) / (b - a);
    f = clamp(f, 0, 1);
    let vv;
    if (space === 'hsl') {
      let h0 = va[0], h1 = vb[0], d = h1 - h0;
      if (d > 0.5) h1 -= 1; else if (d < -0.5) h1 += 1;
      vv = [lerp(h0, h1, f), lerp(va[1], vb[1], f), lerp(va[2], vb[2], f)];
    } else {
      vv = [lerp(va[0], vb[0], f), lerp(va[1], vb[1], f), lerp(va[2], vb[2], f)];
    }
    const rgb = back(vv);
    out[i * 4]     = clamp(Math.round(rgb[0]), 0, 255);
    out[i * 4 + 1] = clamp(Math.round(rgb[1]), 0, 255);
    out[i * 4 + 2] = clamp(Math.round(rgb[2]), 0, 255);
    out[i * 4 + 3] = 255;
  }
  return out;
}
function lutCss(lut) {
  const parts = [];
  for (let i = 0; i <= 32; i++) {
    const j = Math.min(LUT_N - 1, Math.round(i / 32 * LUT_N)) * 4;
    parts.push(rgb2hex(lut[j], lut[j + 1], lut[j + 2]) + ' ' + (i / 32 * 100).toFixed(1) + '%');
  }
  return 'linear-gradient(90deg,' + parts.join(',') + ')';
}
function randomPalette(rnd) {
  rnd = rnd || Math.random;
  const n = 4 + ((rnd() * 3) | 0);
  const baseH = rnd();
  const mode = (rnd() * 3) | 0;                 // 0 analogous 1 triad 2 free
  const stops = [];
  for (let i = 0; i < n; i++) {
    let h;
    if (mode === 0) h = baseH + (i / n) * 0.30 * (rnd() < 0.5 ? 1 : -1);
    else if (mode === 1) h = baseH + (i % 3) / 3 + rnd() * 0.06;
    else h = rnd();
    const t = i / (n - 1);
    const s = 0.45 + rnd() * 0.55;
    const l = clamp(0.06 + Math.pow(t, 0.85) * 0.82 + (rnd() - 0.5) * 0.18, 0.03, 0.97);
    const c = hsl2rgb(h, s, l);
    stops.push({ p: i / n, c: rgb2hex(c[0], c[1], c[2]) });
  }
  if (rnd() < 0.5) stops[0].c = '#04030a';
  return stops;
}

/* =====================================================================
   MODULATION SHAPES (LFOs)
   ---------------------------------------------------------------------
   Every shape is a function of loop position with period exactly 1, and
   every one routes its argument through wrap1() so that the value at
   t=1 is not merely equal to the value at t=0 but produced by the
   identical arithmetic. That is what lets a moving control keep the
   loop exact: render(t, p) is already periodic for fixed p, so if p is
   periodic too then render(1, p(1)) = render(1, p(0)) = render(0, p(0)).
   Rates are whole cycles per loop for the same reason every other rate
   in this program is.
   ===================================================================== */
const wrap1 = x => x - Math.floor(x);

/* deterministic value in [-1,1] from an integer index and a seed */
function ringVal(i, seed) {
  let h = (Math.imul(i | 0, 374761393) + Math.imul(seed | 0, 668265263)) | 0;
  h = Math.imul(h ^ (h >>> 13), 1274126177) | 0;
  h = h ^ (h >>> 16);
  return ((h >>> 0) / 4294967296) * 2 - 1;
}

const LFO_SHAPES = [
  { id: 'sine',  n: 'Sine',           smooth: true },
  { id: 'tri',   n: 'Triangle',       smooth: true },
  { id: 'noise', n: 'Smooth noise',   smooth: true, seeded: true },
  { id: 'drift', n: 'Harmonic drift', smooth: true, seeded: true },
  { id: 'swell', n: 'Swell',          smooth: true, uni: true },
  { id: 'pulse', n: 'Pulse',          smooth: false },
  { id: 'ramp',  n: 'Ramp',           smooth: false }
];
const LFO_BY_ID = {};
LFO_SHAPES.forEach(s => LFO_BY_ID[s.id] = s);

function lfoShape(shape, x, rate, seed) {
  const r = Math.max(1, Math.round(rate || 1));
  switch (shape) {
    case 'tri':   return 1 - 4 * Math.abs(wrap1(r * x + 0.25) - 0.5);
    /* The stepping shapes are offset so their discontinuity never lands on
       the loop boundary. Unshifted, a ramp resets and a pulse flips at
       exactly x=0, which puts the one hard edge in the whole animation on
       the join - it reads as a broken loop rather than as a deliberate
       strobe. Moved into the body of the cycle it reads as intended, and
       the wrap itself stays smooth. */
    case 'ramp':  return 2 * wrap1(r * x + 0.5) - 1;
    case 'pulse': return wrap1(r * x + 0.25) < 0.5 ? 1 : -1;
    case 'swell': { const v = Math.sin(Math.PI * wrap1(r * x)); return v * v; }
    case 'noise': {
      /* value noise around a closed ring of N points, Catmull-Rom through
         them. Closed by construction, so it wanders like noise and still
         arrives back where it began. */
      const N = 2 + 2 * r;
      const p = wrap1(x) * N;
      const i0 = Math.floor(p), f = p - i0;
      const g = k => ringVal(((k % N) + N) % N, seed);
      const a = g(i0 - 1), b = g(i0), c = g(i0 + 1), d = g(i0 + 2);
      const t2 = f * f, t3 = t2 * f;
      const v = 0.5 * (2 * b + (c - a) * f + (2 * a - 5 * b + 4 * c - d) * t2 + (3 * b - a - 3 * c + d) * t3);
      return clamp(v, -1, 1);
    }
    case 'drift': {
      /* 1/f sum over whole harmonics with seeded phases: organic wander,
         still exactly periodic because every harmonic is an integer */
      let sum = 0, norm = 0;
      for (let k = 1; k <= 4; k++) {
        const amp = 1 / k;
        const ph = wrap1(ringVal(k * 17 + seed, seed) * 0.5 + 0.5);
        sum += amp * Math.sin(TAU * wrap1(k * r * x + ph));
        norm += amp;
      }
      return sum / norm;
    }
    default: return Math.sin(TAU * wrap1(r * x));
  }
}
/* Is this modulator continuous all the way round?

   Getting this test right took two wrong attempts, and the reason is worth
   writing down. Comparing lfoValue(m, 0) with lfoValue(m, 1) proves nothing:
   wrap1 sends t=1 to 0, so those are equal by construction. Comparing a hair
   before the wrap proves nothing either, because x = wrap1(t + phase) simply
   keeps advancing through the seam - it is continuous there for ANY rate.

   The discontinuity, when the rate is not a whole number, is not at the seam
   at all. x runs once round the circle over the loop and wraps at t = 1-phase,
   and it is THERE that wrap1(rate*x) jumps. A rate of 1.25 puts a jump of a
   quarter cycle in the middle of the animation. Just as visible, just as
   wrong, simply not where you would think to look for it.

   So the thing to certify is that the shape is continuous across its own wrap
   in x. That is a direct mathematical check, which matters, because an RMS
   pixel comparison saturates on fast-moving content and will happily call a
   modulator that leaps most of its range "seamless". */
function modSeamError(m, mul) {
  const r = (m.rate || 1) * (mul || 1), sd = (m.seed | 0) || 1, sh = m.shape || 'sine';
  return Math.abs(lfoShape(sh, 1 - 1e-7, r, sd) - lfoShape(sh, 0, r, sd));
}
function lfoValue(m, t, mul) {
  /* Wrap t BEFORE adding the phase, not after. wrap1(1) is exactly 0, so the
     argument at t=1 is produced by the identical arithmetic as at t=0 and the
     two agree bit for bit. Doing it the other way round, wrap1(1 + phase) and
     wrap1(0 + phase) differ by one ULP of a double - which sounds harmless
     until Math.round() turns that into a whole step on an integer control, or
     wrap1 lands either side of its own discontinuity. Both were caught
     breaking the seam by tens of levels before this line read like this. */
  /* Speed multiplies the rate. Whole number times whole number is still a
     whole number, so the shape still joins up with itself. */
  return lfoShape(m.shape || 'sine', wrap1(wrap1(t) + (m.phase || 0)),
                  (m.rate || 1) * (mul || 1), (m.seed | 0) || 1);
}
