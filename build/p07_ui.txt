function markDirty() { dirty = true; }
const fmtVal = (v, def) => def.int ? String(Math.round(v)) :
  (Math.abs(v) >= 100 ? v.toFixed(0) : Math.abs(v) >= 10 ? v.toFixed(1) : v.toFixed(3));

/* =====================================================================
   MODULATION UI
   Any control with a path gets a wave button. Turning it on stores an
   LFO against that path and drops an inline panel under the slider; the
   marker on the track then shows where the control actually is on the
   frame being rendered.
   ===================================================================== */
const MOD_RATE  = PI_(0, 'rate', 'Cycles x/loop', 1, 8, 1);
const MOD_DEPTH = P(0, 'depth', 'Depth', 0, 1, 0.25, 0.005);
const MOD_PHASE = P(0, 'phase', 'Phase', 0, 1, 0, 0.005);
const MOD_MASTER_DEF = [P(0, 'depth', 'Master depth', 0, 2, 1, 0.005)];
const MOD_UI = [];

const defaultMod = () => ({ on: true, shape: 'sine', rate: 1, depth: 0.25, phase: 0,
                            seed: 1 + ((Math.random() * 9998) | 0) });

function pathDef(path) {
  const a = path.split(':');
  if (a[0] === 'g') { const g = GEN_BY_ID[a[1]]; return g ? g.params.find(x => x.k === a[2]) : null; }
  const list = { f: FRAME_DEF, p: POST_DEF, x: PAL_EXTRA_DEF, a: ALPHA_DEF }[a[0]];
  return list ? list.find(d => d.k === a[1]) : null;
}
function pathLabel(path) {
  const a = path.split(':'), d = pathDef(path);
  if (a[0] === 'g') { const g = GEN_BY_ID[a[1]]; return (g ? g.name : a[1]) + '  ' + (d ? d.n : a[2]); }
  const grp = { f: 'Framing', p: 'Grade', x: 'Palette', a: 'Alpha' }[a[0]] || a[0];
  return grp + '  ' + (d ? d.n : a[1]);
}
function pruneModUi() {
  for (let i = MOD_UI.length - 1; i >= 0; i--) if (!MOD_UI[i].wrap.isConnected) MOD_UI.splice(i, 1);
}
function refreshModButtons() { MOD_UI.forEach(e => e.wrap._modRefresh && e.wrap._modRefresh()); }
function updateModCount() {
  const n = activeModCount(), tag = $('#modCount');
  if (tag) tag.textContent = n ? n + ' moving' : '';
}
/* live readout: where each animated control actually sits on this frame */
function updateModMarks(t) {
  for (let i = 0; i < MOD_UI.length; i++) {
    const e = MOD_UI[i], m = S.mods[e.path];
    if (!m || !m.on || !S.modMaster.on) continue;
    const v = modulate(e.path, e.get(), t, e.def);
    const span = e.def.mx - e.def.mn;
    e.mark.style.setProperty('--mod', clamp((v - e.def.mn) / (span || 1), 0, 1));
    if (e.liveRef.el) e.liveRef.el.textContent = fmtVal(v, e.def);
  }
}
function buildModPanel(path, def, after) {
  const m = S.mods[path];
  const panel = el('div', 'modpanel');
  const top = el('div', 'mtop');
  const sel = document.createElement('select');
  LFO_SHAPES.forEach(sh => {
    const o = document.createElement('option');
    o.value = sh.id; o.textContent = sh.n + (sh.smooth ? '' : '  (steps)');
    sel.appendChild(o);
  });
  sel.value = m.shape || 'sine';
  const live = el('span', 'live', '');
  live.title = 'the value being rendered on this frame';
  top.appendChild(sel); top.appendChild(live);
  panel.appendChild(top);

  const onCh = () => { markDirty(); if (after) after(); };
  panel.appendChild(mkSlider(MOD_RATE,  () => m.rate,  v => { m.rate = v; buildModList(); }, onCh));
  panel.appendChild(mkSlider(MOD_DEPTH, () => m.depth, v => { m.depth = v; }, onCh));
  panel.appendChild(mkSlider(MOD_PHASE, () => m.phase, v => { m.phase = v; }, onCh));

  const seedRow = el('div', 'seedrow');
  const seedTxt = el('span', '', 'Wander seed ' + m.seed);
  const roll = el('button', 'btn sm', 'roll');
  roll.addEventListener('click', () => {
    m.seed = 1 + ((Math.random() * 9998) | 0);
    seedTxt.textContent = 'Wander seed ' + m.seed; onCh();
  });
  seedRow.appendChild(seedTxt); seedRow.appendChild(roll);
  panel.appendChild(seedRow);

  const jump = el('div', 'jump',
    'This shape steps rather than glides. The loop is still exact, but the jump is visible.');
  panel.appendChild(jump);

  const syncShape = () => {
    const sh = LFO_BY_ID[m.shape] || LFO_SHAPES[0];
    seedRow.style.display = sh.seeded ? '' : 'none';
    jump.style.display = sh.smooth ? 'none' : '';
  };
  sel.addEventListener('change', () => { m.shape = sel.value; syncShape(); onCh(); buildModList(); });
  syncShape();
  return { node: panel, live };
}
function buildModList() {
  const host = $('#modList'); if (!host) return;
  host.innerHTML = '';
  const keys = Object.keys(S.mods).sort();
  if (!keys.length) {
    host.appendChild(el('div', 'note', 'Nothing is animated yet. Press the wave button on any slider.'));
    updateModCount(); return;
  }
  keys.forEach(k => {
    const m = S.mods[k];
    const row = el('div', 'modrow' + (m.on ? '' : ' off'));
    const cb = document.createElement('input');
    cb.type = 'checkbox'; cb.checked = !!m.on;
    cb.addEventListener('change', () => {
      m.on = cb.checked; row.classList.toggle('off', !m.on);
      markDirty(); refreshModButtons(); updateModCount(); pushUrlSoon();
    });
    row.appendChild(cb);
    row.appendChild(el('span', 'nm', pathLabel(k)));
    const sh = LFO_BY_ID[m.shape] || { n: m.shape || '?' };
    row.appendChild(el('span', 'sh', String(sh.n).split(' ')[0].toLowerCase() + ' x' + Math.round(m.rate)));
    const x = el('button', 'x', '\u00d7');
    x.title = 'stop animating this control';
    x.addEventListener('click', () => {
      delete S.mods[k];
      buildModList(); refreshModButtons(); updateModCount(); markDirty(); pushUrlSoon();
    });
    row.appendChild(x);
    host.appendChild(row);
  });
  updateModCount();
}

function mkSlider(def, get, set, after, path) {
  const wrap = el('div', 'ctl' + (def.int ? ' isint' : ''));
  const top = el('div', 'top');
  top.appendChild(el('span', 'name', def.n));
  const canMod = !!path && !def.noMod;
  let modBtn = null;
  if (canMod) {
    modBtn = el('button', 'modbtn');
    modBtn.innerHTML = '&#8767;';
    modBtn.title = 'Animate this control across the loop';
    top.appendChild(modBtn);
  }
  const val = el('span', 'val' + (def.int ? ' int' : ''));
  top.appendChild(val);
  const track = el('div', 'trackwrap');
  const inp = document.createElement('input');
  inp.type = 'range'; inp.min = def.mn; inp.max = def.mx; inp.step = def.st;
  guardSliderTouch(inp);
  const mark = el('i', 'modmark');
  track.appendChild(inp); track.appendChild(mark);
  wrap.appendChild(top); wrap.appendChild(track);

  const show = v => {
    val.textContent = fmtVal(v, def);
    const pct = (v - def.mn) / (def.mx - def.mn) * 100;
    inp.style.setProperty('--pct', clamp(pct, 0, 100) + '%');
  };
  const sync0 = () => { const v = get(); inp.value = v; show(v); };
  inp.addEventListener('input', () => {
    let v = parseFloat(inp.value);
    if (def.int) v = Math.round(v);
    set(v); show(v); markDirty(); if (after) after();
  });
  val.addEventListener('dblclick', () => { set(def.d); sync0(); markDirty(); if (after) after(); });
  val.title = 'double-click to reset';

  let refresh = null;
  if (canMod) {
    const liveRef = { el: null };
    let panel = null;
    refresh = () => {
      const m = S.mods[path];
      modBtn.classList.toggle('on', !!(m && m.on));
      wrap.classList.toggle('modon', !!(m && m.on));
      if (m && !panel) {
        const built = buildModPanel(path, def, after);
        panel = built.node; liveRef.el = built.live;
        wrap.appendChild(panel);
      } else if (!m && panel) {
        panel.remove(); panel = null; liveRef.el = null;
      }
    };
    modBtn.addEventListener('click', () => {
      if (S.mods[path]) delete S.mods[path];
      else S.mods[path] = defaultMod();
      refresh(); markDirty(); buildModList(); updateModCount(); if (after) after();
    });
    wrap._modRefresh = refresh;
    MOD_UI.push({ path, def, wrap, mark, get, liveRef });
    refresh();
  }
  wrap._sync = () => { sync0(); if (refresh) refresh(); };
  sync0();
  return wrap;
}

function buildCtlList(host, defs, store, after, prefix) {
  host.innerHTML = '';
  pruneModUi();
  const syncs = [];
  defs.forEach(d => {
    const s = mkSlider(d, () => store[d.k], v => { store[d.k] = v; }, after, prefix ? prefix + d.k : null);
    host.appendChild(s); syncs.push(s._sync);
  });
  host._syncAll = () => syncs.forEach(f => f());
}

let thumbTimer = 0;
function scheduleThumbs() { clearTimeout(thumbTimer); thumbTimer = setTimeout(renderThumbs, 260); }
function renderThumbs() {
  if (!GL_OK || busy) return;
  const keepGen = S.gen, keepFrame = S.frame, keepPost = S.post, keepBg = S.out.bg;
  const keepMod = S.modMaster.on;
  S.modMaster.on = false;          // a gallery of twitching thumbnails helps nobody
  S.frame = defObj(FRAME_DEF);
  S.post = defObj(POST_DEF);
  S.post.vig = 0; S.post.grain = 0; S.post.chroma = 0;
  S.out.bg = 'opaque';
  const W = 132, H = 44;
  for (const g of GENS) {
    S.gen = g.id;
    if (g.flame) { S.params.flame = Object.assign({}, S.params.flame); }
    const ok = render(0.0, W, H, 1);
    if (!ok) continue;
    const px = readFrame(W, H);
    const c = thumbCanvases[g.id];
    const ctx = c.getContext('2d');
    const img = ctx.createImageData(W, H);
    img.data.set(px);
    ctx.putImageData(img, 0, 0);
  }
  S.gen = keepGen; S.frame = keepFrame; S.post = keepPost; S.out.bg = keepBg;
  S.modMaster.on = keepMod;
  markDirty();
}

/* ---------------- sections ---------------- */
function wireSections() {
  $$('.sec').forEach(sec => {
    const head = sec.querySelector('.head');
    /* Injected markup could carry this class; one missing header used to
       throw here and abort boot, leaving every control on the page dead. */
    if (!head) return;
    head.addEventListener('click', () => {
      sec.classList.toggle('closed');
      try { localStorage.setItem('eloom.sec.' + sec.dataset.sec, sec.classList.contains('closed') ? '1' : '0'); } catch (e) {}
    });
    /* On a phone every open section pushes Export further away, so start
       with only the ones you need and let the rest be tapped open. A stored
       preference always wins. */
    let stored = null;
    try { stored = localStorage.getItem('eloom.sec.' + sec.dataset.sec); } catch (e) {}
    if (stored === '1') sec.classList.add('closed');
    else if (stored === '0') sec.classList.remove('closed');
    else if (isNarrow() && ['gens', 'params', 'out', 'export'].indexOf(sec.dataset.sec) < 0) {
      sec.classList.add('closed');
    }
  });
}

const GIF_BPP = { 256: 4.4, 192: 4.1, 128: 3.7, 96: 3.4, 64: 3.0, 48: 2.7, 32: 2.3, 16: 1.7, 8: 1.1 };
const DITHER_MUL = { fs: 1.0, sierra: 0.95, atkinson: 0.85, bayer8: 0.72, bayer4: 0.68, none: 0.5 };
function updateEstimate() {
  const W = S.out.w, H = S.out.h, N = totalFrames(), px = W * H;
  const bpp = (GIF_BPP[S.gif.colors] || 4) * (DITHER_MUL[S.gif.dither] || 1);
  const gifB = N * px * bpp / 8 + S.gif.colors * 3 + 800;
  const vidB = px * S.out.fps * S.vid.q * S.out.dur / 8;
  const pngB = N * px * 1.6;
  const csTotal = Math.round(S.out.dur * 100);
  const minDelay = Math.floor(csTotal / N);
  const e = $('#est');
  const heavy = GEN_BY_ID[S.gen].heavy;
  const rough = px * S.out.ss * S.out.ss;
  const guessMs = (heavy ? 4.2 : 2.3) * rough / 1e6 + 1.2;   // measured on a mid-range GPU
  const renderS = guessMs * N / 1000;
  let rows = '';
  rows += '<div><span>frames</span><b>' + N + '</b></div>';
  rows += '<div><span>render size</span><b>' + Math.round(W * S.out.ss) + ' x ' + Math.round(H * S.out.ss) + '</b></div>';
  rows += '<div><span>GIF (rough)</span><b class="' + (gifB > 2.5e8 ? 'b' : gifB > 6e7 ? 'w' : '') + '">' + fmtBytes(gifB) + '</b></div>';
  rows += '<div><span>WebM (rough)</span><b>' + fmtBytes(vidB) + '</b></div>';
  rows += '<div><span>PNG zip (rough)</span><b class="' + (pngB > 2e9 ? 'b' : '') + '">' + fmtBytes(pngB) + '</b></div>';
  rows += '<div><span>frame delay</span><b>' + (csTotal / N).toFixed(2) + ' cs</b></div>';
  e.innerHTML = rows;
  let warn = '';
  if (minDelay < 2) warn += 'At ' + S.out.fps + ' fps the GIF frame delay rounds below 2 centiseconds, which many players silently turn into 10 cs. Stay at 50 fps or less. ';
  if (gifB > 1.2e8) warn += 'This GIF will be very large. Drop the colour count, the size, or the frame rate. ';
  if (pngB > 4e9) warn += 'The PNG zip would exceed the 4 GB single-file ZIP limit and will be refused. ';
  if (warn) e.insertAdjacentHTML('beforeend', '<div class="warnbox">' + warn + '</div>');
  e.insertAdjacentHTML('beforeend', '<div class="note">Rough estimate at ~' + fmtTime(renderS) + ' of rendering. Real GIF size is measured while it encodes.</div>');
}

/* ---------------- state <-> url ---------------- */
function snapshot() {
  // only the active pattern's parameters travel: carrying all eighteen would
  // make the shareable link four times longer for nothing
  const p = {}; p[S.gen] = S.params[S.gen];
  // likewise drop modulators belonging to patterns that are not on screen
  const mods = {};
  for (const k in S.mods) {
    if (k.indexOf('g:') === 0 && k.indexOf('g:' + S.gen + ':') !== 0) continue;
    mods[k] = S.mods[k];
  }
  return {
    v: 1, gen: S.gen, seed: S.seed, mods, modMaster: S.modMaster, tempo: S.tempo,
    params: p, frame: S.frame, post: S.post, palX: S.palX, alpha: S.alpha,
    pal: { stops: S.pal.stops, space: S.pal.space, preset: S.pal.preset },
    out: S.out, gif: S.gif, vid: { q: S.vid.q }
  };
}
function applySnapshot(o) {
  if (!o) return;
  if (o.gen && GEN_BY_ID[o.gen]) S.gen = o.gen;
  if (o.seed) S.seed = o.seed;
  if (o.tempo) Object.assign(S.tempo, o.tempo);
  if (o.params) for (const k in o.params) if (S.params[k]) Object.assign(S.params[k], o.params[k]);
  if (o.frame) Object.assign(S.frame, o.frame);
  if (o.post) Object.assign(S.post, o.post);
  if (o.palX) Object.assign(S.palX, o.palX);
  if (o.alpha) Object.assign(S.alpha, o.alpha);
  if (o.pal && o.pal.stops && o.pal.stops.length) {
    S.pal.stops = o.pal.stops.map(s => ({ p: +s.p, c: s.c }));
    S.pal.space = o.pal.space || 'oklab';
    S.pal.preset = (o.pal.preset == null ? -1 : o.pal.preset);
  }
  if (o.out) Object.assign(S.out, o.out);
  if (o.gif) Object.assign(S.gif, o.gif);
  if (o.vid && o.vid.q) S.vid.q = o.vid.q;
  S.mods = o.mods ? JSON.parse(JSON.stringify(o.mods)) : {};
  if (o.modMaster) Object.assign(S.modMaster, o.modMaster);
  refreshAllUI();
}

let urlTimer = 0;
function pushUrlSoon() { clearTimeout(urlTimer); urlTimer = setTimeout(pushUrl, 400); }
function pushUrl() {
  try {
    const json = JSON.stringify(snapshot());
    const b64 = btoa(unescape(encodeURIComponent(json))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    history.replaceState(null, '', '#' + b64);
  } catch (e) {}
  updateEstimate();
}
function readUrl() {
  const h = location.hash.slice(1);
  if (!h) return false;
  try {
    const b64 = h.replace(/-/g, '+').replace(/_/g, '/');
    const json = decodeURIComponent(escape(atob(b64)));
    applySnapshot(JSON.parse(json));
    return true;
  } catch (e) { return false; }
}

/* ---------------- randomize ---------------- */
function randomizeParams(rnd, gid) {
  const g = GEN_BY_ID[gid];
  const st = S.params[gid];
  g.params.forEach(p => {
    if (p.int) {
      // keep motion counts low and interesting
      const span = p.mx - p.mn;
      let v = p.mn + Math.floor(rnd() * (span + 1));
      if (/loop/i.test(p.n) && span > 4) v = p.mn + Math.floor(rnd() * Math.min(4, span + 1));
      st[p.k] = v;
    } else {
      const t = rnd();
      // bias toward the middle so nothing lands on a degenerate extreme
      const b = 0.5 + (t - 0.5) * 0.86;
      st[p.k] = p.mn + b * (p.mx - p.mn);
    }
  });
  if (gid === 'phyllo' && st.n > 200) st.n = 200;

  /* A few patterns have parameter regions that are simply dead, and a blind
     roll lands in them often enough to be annoying. Pull those back. */
  if (g.flame) {
    if (st.q > 5) st.q = 5;
    if (st.it > 30) st.it = 30;
    st.exp = 0.5 + rnd() * 1.1;
    st.gam = 2.0 + rnd() * 1.2;
    st.sym = rnd() < 0.4 ? 1 : 2 + Math.floor(rnd() * 7);
  }
  if (gid === 'julia') {
    /* Julia sets are only interesting for c near the boundary of the
       Mandelbrot set, so walk the main cardioid instead of rolling c
       loose in the plane and landing on a flat wash. */
    const th = rnd() * TAU;
    st.cx = 0.5 * Math.cos(th) - 0.25 * Math.cos(2 * th);
    st.cy = 0.5 * Math.sin(th) - 0.25 * Math.sin(2 * th);
    st.r = 0.02 + rnd() * 0.12;
    st.it = 60 + Math.floor(rnd() * 120);
    st.z = 0.7 + rnd() * 1.1;
  }
}

/* Randomised looks are no use if they come out blown white or nearly black.
   Probe one small frame and pull the exposure back toward a usable mean. */
function balanceExposure() {
  for (let pass = 0; pass < 2; pass++) {
    if (!render(0.3, 64, 36, 1)) return;
    const px = readFrame(64, 36);
    let sum = 0;
    for (let i = 0; i < px.length; i += 4) sum += px[i] * 0.2126 + px[i + 1] * 0.7152 + px[i + 2] * 0.0722;
    const mean = sum / (px.length / 4);
    if (mean > 200) S.post.exposure = clamp(S.post.exposure * (150 / mean), 0.05, 4);
    else if (mean < 26) S.post.exposure = clamp(S.post.exposure * (60 / Math.max(mean, 2)), 0.05, 4);
    else return;
  }
}
function randomizeSpeed(rnd) {
  /* mostly leave it alone; the patterns are authored at speed 1 */
  S.tempo.speed = rnd() < 0.72 ? 1 : 2 + ((rnd() * 2) | 0);
}
function randomizeFrame(rnd) {
  S.frame.zoom = 0.55 + rnd() * 1.6;
  S.frame.rot = (rnd() - 0.5) * 2;
  S.frame.spin = rnd() < 0.55 ? 0 : (rnd() < 0.5 ? 1 : -1);
  S.frame.panx = (rnd() - 0.5) * 0.5;
  S.frame.pany = (rnd() - 0.5) * 0.5;
  S.frame.mirror = rnd() < 0.45 ? [0, 3, 4, 5, 6, 8, 12][(rnd() * 7) | 0] : 0;
  S.frame.mrot = (rnd() - 0.5) * 2;
  S.frame.mspin = rnd() < 0.75 ? 0 : (rnd() < 0.5 ? 1 : -1);
}
function randomizePost(rnd) {
  S.post.bloom = rnd() < 0.25 ? 0 : 0.2 + rnd() * 1.1;
  S.post.thresh = 0.25 + rnd() * 0.5;
  S.post.knee = 0.15 + rnd() * 0.4;
  S.post.radius = 0.8 + rnd() * 2.4;
  S.post.exposure = 0.8 + rnd() * 0.6;
  S.post.contrast = 0.9 + rnd() * 0.5;
  S.post.sat = 0.8 + rnd() * 0.7;
  S.post.gamma = 0.8 + rnd() * 0.5;
  S.post.vig = rnd() < 0.3 ? 0 : rnd() * 0.5;
  S.post.grain = rnd() < 0.5 ? 0 : rnd() * 0.14;
  S.post.chroma = rnd() < 0.5 ? 0 : rnd() * 1.0;
  S.post.posterize = rnd() < 0.85 ? 0 : 3 + ((rnd() * 8) | 0);
  S.post.scan = rnd() < 0.85 ? 0 : rnd() * 0.4;
  S.palX.cycles = rnd() < 0.5 ? 0 : (rnd() < 0.5 ? 1 : -1) * (1 + ((rnd() * 2) | 0));
  S.palX.offset = rnd();
}
/* Roll a small number of controls into motion. Depths stay modest and
   rates stay low: the point is a look that breathes, not one that thrashes. */
function randomizeMods(rnd, replace) {
  if (replace) S.mods = {};
  const g = GEN_BY_ID[S.gen];
  const pool = [];
  g.params.forEach(d => { if (!d.noMod) pool.push(['g:' + g.id + ':' + d.k, d]); });
  ['zoom', 'rot', 'mrot', 'panx', 'pany'].forEach(k => {
    const d = FRAME_DEF.find(x => x.k === k); if (d) pool.push(['f:' + k, d]);
  });
  ['bloom', 'exposure', 'sat', 'contrast', 'chroma', 'vig'].forEach(k => {
    const d = POST_DEF.find(x => x.k === k); if (d) pool.push(['p:' + k, d]);
  });
  const po = PAL_EXTRA_DEF.find(x => x.k === 'offset'); if (po) pool.push(['x:offset', po]);
  const shapes = ['sine', 'sine', 'tri', 'noise', 'noise', 'drift', 'swell'];
  const n = 1 + Math.floor(rnd() * 3);
  for (let i = 0; i < n && pool.length; i++) {
    const j = (rnd() * pool.length) | 0;
    const pick = pool.splice(j, 1)[0];
    S.mods[pick[0]] = {
      on: true,
      shape: shapes[(rnd() * shapes.length) | 0],
      rate: 1 + ((rnd() * 3) | 0),
      depth: 0.05 + rnd() * 0.3,
      phase: rnd(),
      seed: 1 + ((rnd() * 9998) | 0)
    };
  }
}

function doRandomize(full) {
  S.seed = randSeedWord();
  const rnd = mulberry32(hashStr(S.seed));
  if (full) {
    const pick = GENS.filter(g => !g.flame || Math.random() < 0.25);
    S.gen = pick[(rnd() * pick.length) | 0].id;
    const pi = (rnd() * PAL_PRESETS.length) | 0;
    if (rnd() < 0.6) {
      S.pal.stops = PAL_PRESETS[pi].s.map((c, j) => ({ p: j / PAL_PRESETS[pi].s.length, c }));
      S.pal.preset = pi;
    } else {
      S.pal.stops = randomPalette(rnd); S.pal.preset = -1;
    }
    randomizeFrame(rnd);
    randomizePost(rnd);
    randomizeSpeed(rnd);
    randomizeMods(rnd, true);
  } else {
    S.mods = {};
    S.tempo.speed = 1;
  }
  randomizeParams(rnd, S.gen);
  balanceExposure();
  refreshAllUI(); pushUrl(); scheduleThumbs();
}

/* ---------------- layout ---------------- */
function layout() {
  const wrap = $('.canvasWrap');
  const r = wrap.getBoundingClientRect();
  const availW = Math.max(80, r.width - 32), availH = Math.max(80, r.height - 32);
  const ar = S.out.w / S.out.h;
  let w = availW, h = w / ar;
  if (h > availH) { h = availH; w = h * ar; }
  const f = $('#viewFrame');
  f.style.width = Math.round(w) + 'px';
  f.style.height = Math.round(h) + 'px';
  cvs.style.width = Math.round(w) + 'px';
  cvs.style.height = Math.round(h) + 'px';
}

/* ---------------- playback ---------------- */
let fpsAvg = 0;
const isNarrow = () => window.innerWidth <= 1040;
function previewDims() {
  const q = parseFloat($('#previewQ').value) || 1;
  let w = Math.max(64, Math.round(S.out.w * q)), h = Math.max(36, Math.round(S.out.h * q));
  /* A phone has no business rendering 2560x1440 twice over for a preview
     that is displayed 360 pixels wide. */
  const cap = isNarrow() ? 800 : 1600;
  if (Math.max(w, h) > cap) { const k = cap / Math.max(w, h); w = Math.round(w * k); h = Math.round(h * k); }
  return [w, h];
}
function tick(ts) {
  requestAnimationFrame(tick);
  if (busy) { lastRAF = ts; return; }
  const dt = lastRAF ? (ts - lastRAF) / 1000 : 0;
  lastRAF = ts;
  /* theatre mode owns the canvas and its own render schedule; the normal
     preview path is skipped entirely while it is up */
  if (TH.active) { theatreFrame(Math.min(dt, 0.1)); return; }
  const N = totalFrames();
  if (seamMode) {
    seamTick += dt;
    const phase = Math.floor(seamTick * 2) % 2;
    const want = phase === 0 ? 0 : (N - 1) / N;
    if (Math.abs(tNow - want) > 1e-9) { tNow = want; dirty = true; }
    $('#seamBadge').className = 'show';
    $('#seamBadge').textContent = phase === 0 ? 'seam check: frame 0' : 'seam check: frame ' + (N - 1);
  } else if (playing) {
    tNow = (tNow + dt / Math.max(S.out.dur, 0.01)) % 1;
    dirty = true;
  }
  if (!dirty) return;
  dirty = false;
  const [pw, ph] = previewDims();
  const t0 = performance.now();
  const ok = render(tNow, pw, ph, S.out.ss);
  const ms = performance.now() - t0;
  fpsAvg = fpsAvg ? fpsAvg * 0.9 + ms * 0.1 : ms;
  if (!ok) {
    showOverlay('Shader problem', lastRenderErr + '\nOpen the console for the full compile log.');
  }
  updateModMarks(tNow);
  const fi = Math.floor(tNow * N + 1e-6) % N;
  $('#tStat').innerHTML = 'frame <b>' + fi + '</b>/' + N;
  const scr = $('#scrub');
  if (document.activeElement !== scr) {
    scr.value = Math.round(tNow * 1000);
    scr.style.setProperty('--pct', (tNow * 100).toFixed(1) + '%');
  }
  const f = 1000 / Math.max(fpsAvg, 0.01);
  const fs = $('#fpsStat');
  fs.textContent = f > 999 ? '999+' : f.toFixed(0);
  fs.className = 'fps' + (f < 12 ? ' bad' : f < 26 ? ' slow' : '');
}
