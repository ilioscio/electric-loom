/* =====================================================================
   SPEED
   ---------------------------------------------------------------------
   A whole-number multiplier on every per-loop rate in the look. Whole
   times whole is whole, so every coefficient on TA stays an integer and
   the loop stays exact - which is the only reason this control can exist
   at all. It cannot go below 1: halving a rate of 1 would give 0.5 and
   break the rule. To slow a look down, lengthen the loop instead.

   Which controls count as rates is taken from their label: every temporal
   control in this program is named "... x/loop" and no structural one is.
   That convention is load-bearing now, so keep to it.
   ===================================================================== */
const SPEED_DEF = { i: 0, k: 'speed', n: 'Speed', mn: 1, mx: 6, d: 1, st: 1, int: true, noMod: true };
const isTempo = d => /x\/loop/.test(d.n);

/* =====================================================================
   MODULATION
   ---------------------------------------------------------------------
   Any control can be handed an LFO. A modulator is stored against a
   stable path, and every read of a control in the render pass goes
   through modulate() instead of touching the store directly.

   Why this cannot break the loop: render(t, p) is already exactly
   periodic in t for any fixed p, and every LFO is exactly periodic in t,
   so render(1, p(1)) = render(1, p(0)) = render(0, p(0)). The one thing
   that has to be watched is precision - wrap1(1 + phase) and
   wrap1(0 + phase) can differ in the last bit of a double - so the
   result is rounded to float32 here, which is the width it is uploaded
   at anyway. The two ends of the loop then agree bit for bit.

   Paths:  g:<generator>:<key>   f:<key> framing   p:<key> grade
           x:<key> palette       a:<key> alpha
   ===================================================================== */
S.mods = {};
S.modMaster = { on: true, depth: 1 };

/* Rates are multiplied by Speed, then clamped to the range the pattern was
   authored with. The clamp matters: past a certain number of cycles per loop
   the motion aliases against the frame rate and reads as strobing rather than
   as speed, and the authored maximum is where that was judged. */
function applySpeed(v, def) {
  const s = S.tempo.speed | 0;
  if (!def || !def.tempo || s <= 1) return v;
  return clamp(Math.round(v * s), def.mn, def.mx);
}
function modulate(path, base, t, def) {
  const m = S.mods[path];
  if (!m || !m.on || !S.modMaster.on || !m.depth) return applySpeed(base, def);
  const range = def ? (def.mx - def.mn) : 1;
  let v = base + lfoValue(m, t, S.tempo.speed) * m.depth * S.modMaster.depth * range;
  if (def) v = clamp(v, def.mn, def.mx);
  v = Math.fround(v);
  if (def && def.int) v = Math.round(v);
  return applySpeed(v, def);
}
function resolveGroup(prefix, defs, store, t) {
  const o = {};
  for (let i = 0; i < defs.length; i++) {
    const d = defs[i];
    o[d.k] = modulate(prefix + d.k, store[d.k], t, d);
  }
  return o;
}
const resolvedFrame = t => resolveGroup('f:', FRAME_DEF, S.frame, t);
const resolvedPost  = t => resolveGroup('p:', POST_DEF,  S.post,  t);
const resolvedPalX  = t => resolveGroup('x:', PAL_EXTRA_DEF, S.palX, t);
const resolvedAlpha = t => resolveGroup('a:', ALPHA_DEF, S.alpha, t);
function resolvedParams(g, t) {
  const pre = 'g:' + g.id + ':', store = S.params[g.id], o = {};
  for (let i = 0; i < g.params.length; i++) {
    const d = g.params[i];
    o[d.k] = modulate(pre + d.k, store[d.k], t, d);
  }
  return o;
}
function activeModCount() {
  let n = 0;
  for (const k in S.mods) if (S.mods[k] && S.mods[k].on) n++;
  return n;
}
/* worst offender among the live modulators, and how badly it misses */
function worstModSeam() {
  let worst = 0, where = '';
  for (const k in S.mods) {
    const m = S.mods[k];
    if (!m || !m.on) continue;
    const e = modSeamError(m, S.tempo.speed);
    if (e > worst) { worst = e; where = k; }
  }
  return { err: worst, path: where };
}

let GL_OK = !!gl;
let FLOAT_OK = false, FLOAT_BLEND = false, COLOR_FMT = 0;

if (GL_OK) {
  const cbf = gl.getExtension('EXT_color_buffer_float');
  const cbh = gl.getExtension('EXT_color_buffer_half_float');
  gl.getExtension('OES_texture_float_linear');
  const fb = gl.getExtension('EXT_float_blend');

  FLOAT_BLEND = !!fb;

const progCache = new Map();
function makeShader(type, src) {
  const s = gl.createShader(type);
  gl.shaderSource(s, src); gl.compileShader(s);
  if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) {
    const log = gl.getShaderInfoLog(s);
    console.error('shader compile failed:\n' + log + '\n---\n' +
      src.split('\n').map((l, i) => (i + 1) + ': ' + l).join('\n'));
    gl.deleteShader(s); return null;
  }
  return s;
}
function makeProg(key, vsSrc, fsSrc) {
  if (progCache.has(key)) return progCache.get(key);
  const vs = makeShader(gl.VERTEX_SHADER, vsSrc);
  const fs = makeShader(gl.FRAGMENT_SHADER, fsSrc);
  if (!vs || !fs) { progCache.set(key, null); return null; }
  const p = gl.createProgram();
  gl.attachShader(p, vs); gl.attachShader(p, fs); gl.linkProgram(p);
  if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
    console.error('link failed: ' + gl.getProgramInfoLog(p));
    progCache.set(key, null); return null;
  }
  gl.deleteShader(vs); gl.deleteShader(fs);
  const uni = {};
  const n = gl.getProgramParameter(p, gl.ACTIVE_UNIFORMS);
  for (let i = 0; i < n; i++) {
    const info = gl.getActiveUniform(p, i);
    const nm = info.name.replace(/\[0\]$/, '');
    uni[nm] = gl.getUniformLocation(p, info.name);
  }
  const rec = { p, uni };
  progCache.set(key, rec);
  return rec;
}
function genProg(g) {
  return makeProg('gen:' + g.id, VS_QUAD, GLSL_PRELUDE + g.body + GLSL_MAIN);
}

function u1f(r, n, v) { if (r.uni[n] !== undefined) gl.uniform1f(r.uni[n], v); }
function u2f(r, n, a, b) { if (r.uni[n] !== undefined) gl.uniform2f(r.uni[n], a, b); }
function u1i(r, n, v) { if (r.uni[n] !== undefined) gl.uniform1i(r.uni[n], v); }
function u1fv(r, n, v) { if (r.uni[n] !== undefined) gl.uniform1fv(r.uni[n], v); }

/* ---- render targets ---- */
const SCRATCH_UNIT = 7;   // never touch units 0-2, the render passes own those
function makeFbo(w, h, fmt) {
  const tex = gl.createTexture();
  gl.activeTexture(gl.TEXTURE0 + SCRATCH_UNIT);
  gl.bindTexture(gl.TEXTURE_2D, tex);
  const internal = fmt || COLOR_FMT;
  const type = internal === gl.RGBA8 ? gl.UNSIGNED_BYTE : gl.HALF_FLOAT;
  gl.texImage2D(gl.TEXTURE_2D, 0, internal, w, h, 0, gl.RGBA, type, null);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
  const fb = gl.createFramebuffer();
  gl.bindFramebuffer(gl.FRAMEBUFFER, fb);
  gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, tex, 0);
  gl.bindFramebuffer(gl.FRAMEBUFFER, null);
  gl.bindTexture(gl.TEXTURE_2D, null);
  gl.activeTexture(gl.TEXTURE0);
  return { fb, tex, w, h, fmt: internal };
}
const rt = {};
function need(name, w, h, fmt) {
  const cur = rt[name];
  if (cur && cur.w === w && cur.h === h && (!fmt || cur.fmt === fmt)) return cur;
  if (cur) { gl.deleteTexture(cur.tex); gl.deleteFramebuffer(cur.fb); }
  return (rt[name] = makeFbo(w, h, fmt));
}

/* ---- palette texture ---- */
let palTex = null;
function uploadPalette() {
  if (!GL_OK) return;
  if (!palTex) {
    palTex = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, palTex);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.REPEAT);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
  }
  gl.bindTexture(gl.TEXTURE_2D, palTex);
  gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA8, LUT_N, 1, 0, gl.RGBA, gl.UNSIGNED_BYTE, LUT);
}

let vao = null;
function ensureVao() { if (!vao) { vao = gl.createVertexArray(); } gl.bindVertexArray(vao); }
function drawQuad() { gl.drawArrays(gl.TRIANGLES, 0, 3); }

/* ---- common per-frame uniform values ----
   Spin counts are whole turns per loop, so only the fractional turn matters.
   Keeping angles inside one turn is mathematically identical and stops the
   mantissa draining as the spin count climbs, which is what used to make
   hard-edged patterns twitch at the seam. (wrap1 lives up in the LFO
   section - it is the same idea applied to the same problem.)
   A modulated spin count still lands on a whole number at t=1, because
   modulate() rounds integer controls, so wrap1(t*spin) is still 0 at both
   ends of the loop. */
function frameUniforms(t) {
  const F = resolvedFrame(t), X = resolvedPalX(t);
  return {
    rot: F.rot + TAU * wrap1(t * F.spin),
    zoom: F.zoom,
    pan: [F.panx, F.pany],
    mirror: F.mirror,
    mirrorRot: F.mrot + TAU * wrap1(t * F.mspin),
    palRot: X.offset + wrap1(t * X.cycles),
    seedF: (hashStr(S.seed) % 100000) / 1000
  };
}

const POINTS_PER_PASS = 262144;

/* =====================================================================
   THE RENDERER
   render(t, W, H, ss) -> leaves the finished RGBA frame in the canvas
   drawing buffer (W x H). Deterministic: same inputs, same pixels.
   ===================================================================== */
let lastRenderErr = '';
/* Theatre mode borrows the renderer to capture a look into an offscreen
   texture instead of the canvas, so two looks can be composited by a
   transition shader. When RENDER_TARGET is a { fb, w, h } record the final
   composite is drawn there; left null it draws to the canvas exactly as
   before. Nothing but theatre touches this, and it always resets it to null
   after use. */
let RENDER_TARGET = null;
function render(t, W, H, ss) {
  if (!GL_OK) return false;
  ensureVao();
  const rw = Math.max(1, Math.round(W * ss)), rh = Math.max(1, Math.round(H * ss));
  const g = GEN_BY_ID[S.gen];
  const fu = frameUniforms(t);
  const AL = resolvedAlpha(t);
  const alphaOn = S.out.bg === 'alpha';
  const scene = need('scene', rw, rh);

  gl.disable(gl.BLEND);
  gl.disable(gl.DEPTH_TEST);
  gl.activeTexture(gl.TEXTURE0);
  gl.bindTexture(gl.TEXTURE_2D, palTex);

  if (g.flame) {
    const pv = resolvedParams(g, t);
    const acc = need('acc', rw, rh, FLOAT_OK ? COLOR_FMT : gl.RGBA8);
    const pa = makeProg('flame', FLAME_VS, FLAME_FS);
    if (!pa) { lastRenderErr = 'flame shader failed'; return false; }
    gl.bindFramebuffer(gl.FRAMEBUFFER, acc.fb);
    gl.viewport(0, 0, rw, rh);
    gl.clearColor(0, 0, 0, 0); gl.clear(gl.COLOR_BUFFER_BIT);
    gl.enable(gl.BLEND);
    gl.blendFunc(gl.ONE, gl.ONE);
    gl.useProgram(pa.p);
    gl.activeTexture(gl.TEXTURE0);
    gl.bindTexture(gl.TEXTURE_2D, palTex);
    u2f(pa, 'uRes', rw, rh);
    u1f(pa, 'uT', t); u1f(pa, 'uSeed', fu.seedF);
    u1f(pa, 'uNT', pv.nt); u1f(pa, 'uVarA', pv.va); u1f(pa, 'uVarB', pv.vb);
    u1f(pa, 'uMix', pv.mix); u1f(pa, 'uSym', pv.sym); u1f(pa, 'uSpread', pv.spread);
    u1f(pa, 'uJitter', pv.jit); u1f(pa, 'uSpeed', pv.sp); u1f(pa, 'uIter', pv.it);
    u1f(pa, 'uPointSize', pv.ps * ss);
    u1f(pa, 'uZoom', fu.zoom); u1f(pa, 'uRot', fu.rot);
    u2f(pa, 'uPan', fu.pan[0], fu.pan[1]);
    u1f(pa, 'uPalRot', fu.palRot);
    u1i(pa, 'uPal', 0);
    u1f(pa, 'uWeight', 1.0);
    const passes = Math.max(1, Math.round(pv.q));
    for (let i = 0; i < passes; i++) {
      u1f(pa, 'uPass', i);
      gl.drawArrays(gl.POINTS, 0, POINTS_PER_PASS);
    }
    gl.disable(gl.BLEND);

    const pr = makeProg('flameres', VS_QUAD, FLAME_RESOLVE);
    if (!pr) { lastRenderErr = 'flame resolve failed'; return false; }
    const ptsz = pv.ps * ss;
    /* Reference density for the log tone map. A flame's histogram is wildly
       peaked - the core can run a hundred times the mean - so normalising to
       anything near the average clips it to white. This reference is fixed
       (never a per-frame maximum, which would make the brightness pump from
       frame to frame and break the loop) and generous enough to leave the
       core headroom; Exposure is then a straightforward brightness dial. */
    const refDens = Math.max(1e-3, (POINTS_PER_PASS * passes) / (rw * rh) * ptsz * ptsz * 34.0);
    gl.bindFramebuffer(gl.FRAMEBUFFER, scene.fb);
    gl.viewport(0, 0, rw, rh);
    gl.useProgram(pr.p);
    gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, acc.tex);
    u1i(pr, 'uAcc', 0);
    u2f(pr, 'uRes', rw, rh);
    u1f(pr, 'uExposure', pv.exp); u1f(pr, 'uGamma', pv.gam); u1f(pr, 'uSat', pv.sat);
    u1f(pr, 'uNorm', Math.log(1 + refDens));
    u1f(pr, 'uMirror', fu.mirror); u1f(pr, 'uMirrorRot', fu.mirrorRot);
    u1f(pr, 'uAlphaMode', alphaOn ? 1 : 0);
    u1f(pr, 'uAlphaGain', AL.alphaGain); u1f(pr, 'uAlphaBias', AL.alphaBias);
    drawQuad();
  } else {
    const pg = genProg(g);
    if (!pg) { lastRenderErr = 'generator "' + g.name + '" failed to compile'; return false; }
    const arr = new Float32Array(16);
    const rp = resolvedParams(g, t);
    g.params.forEach(pd => { arr[pd.i] = rp[pd.k]; });
    gl.bindFramebuffer(gl.FRAMEBUFFER, scene.fb);
    gl.viewport(0, 0, rw, rh);
    gl.useProgram(pg.p);
    gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, palTex);
    u1i(pg, 'uPal', 0);
    u2f(pg, 'uRes', rw, rh);
    u1f(pg, 'uT', t); u1f(pg, 'uSeed', fu.seedF);
    u1fv(pg, 'uP', arr);
    u1f(pg, 'uZoom', fu.zoom); u1f(pg, 'uRot', fu.rot);
    u1f(pg, 'uMirror', fu.mirror); u1f(pg, 'uMirrorRot', fu.mirrorRot);
    u1f(pg, 'uPalRot', fu.palRot);
    u2f(pg, 'uPan', fu.pan[0], fu.pan[1]);
    u1f(pg, 'uAlphaMode', alphaOn ? 1 : 0);
    u1f(pg, 'uAlphaGain', AL.alphaGain);
    u1f(pg, 'uAlphaBias', AL.alphaBias);
    drawQuad();
  }

  /* ---- bloom ---- */
  const P_ = resolvedPost(t);
  const bw = Math.max(2, rw >> 2), bh = Math.max(2, rh >> 2);
  const b0 = need('b0', bw, bh), b1 = need('b1', bw, bh), b2 = need('b2', bw, bh), b3 = need('b3', bw, bh);
  if (P_.bloom > 0.0005) {
    const pb = makeProg('bright', VS_QUAD, FS_BRIGHT);
    const pblur = makeProg('blur', VS_QUAD, FS_BLUR);
    gl.bindFramebuffer(gl.FRAMEBUFFER, b0.fb); gl.viewport(0, 0, bw, bh);
    gl.useProgram(pb.p);
    gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, scene.tex);
    u1i(pb, 'uSrc', 0); u2f(pb, 'uRes', bw, bh);
    u1f(pb, 'uThresh', P_.thresh); u1f(pb, 'uKnee', P_.knee);
    drawQuad();
    const r = P_.radius;
    const chain = [[b0, b1, r, 0], [b1, b2, 0, r], [b2, b3, r * 2.7, 0], [b3, b1, 0, r * 2.7]];
    gl.useProgram(pblur.p); u1i(pblur, 'uSrc', 0);
    for (const [src, dst, dx, dy] of chain) {
      gl.bindFramebuffer(gl.FRAMEBUFFER, dst.fb); gl.viewport(0, 0, bw, bh);
      gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, src.tex);
      u2f(pblur, 'uRes', bw, bh); u2f(pblur, 'uDir', dx, dy);
      drawQuad();
    }
    // b2 holds the first blur level, b1 the wide one
  }

  /* ---- composite to the canvas (or an offscreen target for theatre) ---- */
  const tgt = RENDER_TARGET;
  if (!tgt && (cvs.width !== W || cvs.height !== H)) { cvs.width = W; cvs.height = H; }
  const pc = makeProg('comp', VS_QUAD, FS_COMPOSITE);
  gl.bindFramebuffer(gl.FRAMEBUFFER, tgt ? tgt.fb : null);
  gl.viewport(0, 0, W, H);
  gl.useProgram(pc.p);
  gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, scene.tex); u1i(pc, 'uScene', 0);
  gl.activeTexture(gl.TEXTURE1); gl.bindTexture(gl.TEXTURE_2D, (P_.bloom > 0.0005 ? b2 : b0).tex); u1i(pc, 'uBloomA', 1);
  gl.activeTexture(gl.TEXTURE2); gl.bindTexture(gl.TEXTURE_2D, (P_.bloom > 0.0005 ? b1 : b0).tex); u1i(pc, 'uBloomB', 2);
  u2f(pc, 'uRes', W, H);
  u1f(pc, 'uBloom', P_.bloom); u1f(pc, 'uExposure', P_.exposure);
  u1f(pc, 'uContrast', P_.contrast); u1f(pc, 'uSat', P_.sat); u1f(pc, 'uGammaC', P_.gamma);
  u1f(pc, 'uVig', P_.vig); u1f(pc, 'uVigSoft', P_.vigSoft);
  u1f(pc, 'uGrain', P_.grain); u1f(pc, 'uChroma', P_.chroma);
  u1f(pc, 'uPosterize', P_.posterize); u1f(pc, 'uScan', P_.scan); u1f(pc, 'uScanFreq', P_.scanFreq);
  const _N = totalFrames();
  u1f(pc, 'uFrameIdx', ((Math.floor(t * _N + 1e-6) % _N) + _N) % _N);   // grain repeats with the loop
  u1f(pc, 'uAlphaOut', alphaOn ? 1 : 0);
  gl.activeTexture(gl.TEXTURE0);
  drawQuad();
  return true;
}

/* read the canvas back, flipped to top-down RGBA */
let readBuf = null;
function readFrame(W, H) {
  const need = W * H * 4;
  if (!readBuf || readBuf.length !== need) readBuf = new Uint8Array(need);
  gl.bindFramebuffer(gl.FRAMEBUFFER, null);
  gl.readPixels(0, 0, W, H, gl.RGBA, gl.UNSIGNED_BYTE, readBuf);
  const out = new Uint8Array(need);
  const row = W * 4;
  for (let y = 0; y < H; y++) {
    out.set(readBuf.subarray((H - 1 - y) * row, (H - y) * row), y * row);
  }
  return out;
}
