const PRESET_KEY = 'eloom.presets.v1';
function loadPresets() { try { return JSON.parse(localStorage.getItem(PRESET_KEY) || '[]'); } catch (e) { return []; } }
function savePresets(list) { try { localStorage.setItem(PRESET_KEY, JSON.stringify(list)); } catch (e) { toast('Could not save: local storage is full or blocked.'); } }
function buildPresetList() {
  const host = $('#presetList'); host.innerHTML = '';
  const list = loadPresets();
  if (!list.length) { host.appendChild(el('div', 'note', 'Nothing saved yet. Name a look and hit Save.')); return; }
  list.forEach((p, i) => {
    const row = el('div', 'row');
    const b = el('button', 'btn sm', p.name);
    b.style.flex = '1'; b.style.justifyContent = 'flex-start';
    b.addEventListener('click', () => { applySnapshot(p.data); pushUrl(); scheduleThumbs(); toast('Loaded "' + p.name + '"'); });
    const x = el('button', 'btn sm danger', '×');
    x.addEventListener('click', () => { const l = loadPresets(); l.splice(i, 1); savePresets(l); buildPresetList(); });
    row.appendChild(b); row.appendChild(x);
    host.appendChild(row);
  });
}

/* =====================================================================
   TIP LINK
   ---------------------------------------------------------------------
   Configured at build time as three meta tags, so it costs nothing: a
   plain anchor needs no third-party script and no CSP allowance, unlike
   an ad unit. Absent from the default build entirely.
   ===================================================================== */
const TIP = { url: "", label: "", note: "" };
const TIP_KEY = 'eloom.tip.hidden';
const TIP_QUIET_DAYS = 60;
let tipShownThisSession = false;

function readTipConfig() {
  const meta = n => { const m = document.querySelector('meta[name="el-tip-' + n + '"]'); return m ? m.content.trim() : ''; };
  const url = meta('url');
  /* An operator typo must not become a javascript: link in everyone's
     browser, so only real web URLs are accepted. */
  if (!/^https?:\/\//i.test(url)) return;
  TIP.url = url;
  TIP.label = meta('label') || 'Tip the developer';
  TIP.note = meta('note');
  const btn = $('#tipBtn');
  btn.href = TIP.url;
  btn.textContent = TIP.label;
  btn.title = TIP.note || TIP.label;
  btn.style.display = '';
}
function tipIsMuted() {
  try {
    const t = parseInt(localStorage.getItem(TIP_KEY) || '0', 10);
    return t > 0 && (Date.now() - t) < TIP_QUIET_DAYS * 86400000;
  } catch (e) { return false; }
}
/* Shown once per session, after an export has actually landed, and only
   until it is waved away. A creative tool that nags is a worse tool. */
function maybeShowTip() {
  if (!TIP.url || tipShownThisSession || tipIsMuted()) return;
  tipShownThisSession = true;
  const card = el('div', 'tipcard');
  const h = el('h4', '', 'Rendered on your machine, free');
  const p = el('p', '', TIP.note ||
    'Nothing you make here is uploaded and nothing is watermarked. If it saved you some time, you can throw the developer a few quid.');
  const row = el('div', 'row');
  const go = document.createElement('a');
  go.className = 'btn sm tip'; go.href = TIP.url; go.target = '_blank';
  go.rel = 'noopener noreferrer'; go.textContent = TIP.label;
  go.style.flex = '1';
  const no = el('button', 'btn sm', 'Not now');
  no.addEventListener('click', () => {
    try { localStorage.setItem(TIP_KEY, String(Date.now())); } catch (e) {}
    card.remove();
  });
  row.appendChild(go); row.appendChild(no);
  card.appendChild(h); card.appendChild(p); card.appendChild(row);
  $('#outList').appendChild(card);
}

/* =====================================================================
   WIRING
   ===================================================================== */
function wire() {
  /* framing controls, with the seed on top */
  const fc = $('#frameCtls');
  buildCtlList(fc, FRAME_DEF, S.frame, pushUrlSoon, 'f:');
  const seedWrap = el('div', 'ctl');
  const st = el('div', 'top'); st.appendChild(el('span', 'name', 'Seed'));
  seedWrap.appendChild(st);
  const sr = el('div', 'seedRow');
  const si = document.createElement('input'); si.type = 'text'; si.id = 'seedInput'; si.value = S.seed;
  si.addEventListener('input', () => { S.seed = si.value || 'x'; markDirty(); pushUrlSoon(); scheduleThumbs(); });
  const sb = el('button', 'btn sm', 'roll');
  sb.addEventListener('click', () => { S.seed = randSeedWord(); si.value = S.seed; markDirty(); pushUrl(); scheduleThumbs(); });
  sr.appendChild(si); sr.appendChild(sb);
  seedWrap.appendChild(sr);
  seedWrap.appendChild(el('div', 'hint', 'Only the patterns with scattered elements use it: metaballs, voronoi, truchet, hex, ripples, flame.'));
  fc.insertBefore(seedWrap, fc.firstChild);

  buildCtlList($('#postCtls'), POST_DEF, S.post, pushUrlSoon, 'p:');
  buildCtlList($('#palExtra'), PAL_EXTRA_DEF, S.palX, pushUrlSoon, 'x:');
  buildCtlList($('#alphaCtls'), ALPHA_DEF, S.alpha, pushUrlSoon, 'a:');
  buildCtlList($('#modDepthCtl'), MOD_MASTER_DEF, S.modMaster, pushUrlSoon);
  buildCtlList($('#ditherStrCtl'), [P(0, 'strength', 'Dither strength', 0, 1, 0.9, 0.005)], S.gif, () => { updateEstimate(); pushUrlSoon(); });

  $('#modMaster').addEventListener('change', e => {
    S.modMaster.on = e.target.checked;
    refreshModButtons(); updateModCount(); markDirty(); pushUrlSoon();
  });
  $('#btnModRandom').addEventListener('click', () => {
    randomizeMods(Math.random, true);
    refreshModButtons(); buildModList(); markDirty(); pushUrl();
    toast(activeModCount() + ' control(s) set moving.');
  });
  $('#btnModClear').addEventListener('click', () => {
    S.mods = {};
    refreshModButtons(); buildModList(); markDirty(); pushUrl();
  });

  $('#palSpace').addEventListener('change', e => { S.pal.space = e.target.value; rebuildLut(); pushUrl(); });
  $('#btnAddStop').addEventListener('click', () => {
    const n = S.pal.stops.length;
    S.pal.stops.push({ p: clamp(n / (n + 1), 0, 0.999), c: '#ffffff' });
    S.pal.preset = -1; buildStops(); rebuildLut(); pushUrl();
  });
  $('#btnRevPal').addEventListener('click', () => {
    S.pal.stops = S.pal.stops.slice().reverse().map((s, i, a) => ({ p: i / a.length, c: s.c }));
    S.pal.preset = -1; buildStops(); rebuildLut(); pushUrl();
  });
  $('#btnRandPal').addEventListener('click', () => {
    S.pal.stops = randomPalette(); S.pal.preset = -1;
    buildStops(); buildPalPresets(); rebuildLut(); pushUrl();
  });

  /* output */
  const setSize = (w, h) => {
    S.out.w = clamp(Math.round(w / 2) * 2, 16, 4096);
    S.out.h = clamp(Math.round(h / 2) * 2, 16, 4096);
    $('#outW').value = S.out.w; $('#outH').value = S.out.h;
    syncSizePreset(); updateEstimate(); layout(); markDirty(); pushUrlSoon();
  };
  $('#sizePreset').addEventListener('change', e => {
    if (e.target.value === 'custom') return;
    const [w, h] = e.target.value.split('x').map(Number);
    setSize(w, h);
  });
  $('#outW').addEventListener('change', e => setSize(+e.target.value, S.out.h));
  $('#outH').addEventListener('change', e => setSize(S.out.w, +e.target.value));
  $('#outDur').addEventListener('change', e => {
    S.out.dur = clamp(parseFloat(e.target.value) || 4, 0.5, 15);
    e.target.value = S.out.dur; updateEstimate(); markDirty(); pushUrlSoon();
  });
  $('#outFps').addEventListener('change', e => { S.out.fps = +e.target.value; updateEstimate(); markDirty(); pushUrlSoon(); });
  $('#outSS').addEventListener('change', e => { S.out.ss = +e.target.value; updateEstimate(); markDirty(); pushUrlSoon(); });
  $('#bgMode').addEventListener('change', e => {
    S.out.bg = e.target.value;
    $('#alphaCtls').style.display = S.out.bg === 'alpha' ? '' : 'none';
    markDirty(); pushUrlSoon();
  });
  $('#gifColors').addEventListener('change', e => { S.gif.colors = +e.target.value; updateEstimate(); pushUrlSoon(); });
  $('#gifDither').addEventListener('change', e => { S.gif.dither = e.target.value; updateEstimate(); pushUrlSoon(); });
  $('#gifSharpPal').addEventListener('change', e => { S.gif.kmeans = e.target.checked; pushUrlSoon(); });
  $('#vidCodec').addEventListener('change', e => { S.vid.codec = e.target.value; pushUrlSoon(); });
  $('#vidQ').addEventListener('change', e => { S.vid.q = parseFloat(e.target.value); updateEstimate(); pushUrlSoon(); });

  /* transport */
  $('#btnPlay').addEventListener('click', () => togglePlay());
  $('#btnSeam').addEventListener('click', () => toggleSeam());
  $('#scrub').addEventListener('input', e => {
    tNow = clamp(+e.target.value / 1000, 0, 0.9999);
    e.target.style.setProperty('--pct', (tNow * 100).toFixed(1) + '%');
    markDirty();
  });
  $('#previewQ').addEventListener('change', markDirty);
  $('#btnVerify').addEventListener('click', verifyLoop);

  /* export */
  $('#btnGif').addEventListener('click', exportGif);
  $('#btnWebm').addEventListener('click', exportWebm);
  $('#btnPngs').addEventListener('click', exportPngs);
  $('#btnCancel').addEventListener('click', () => { CANCEL = true; toast('Stopping...'); });

  /* header */
  $('#btnRandom').addEventListener('click', () => doRandomize(true));
  $('#btnRandParams').addEventListener('click', () => doRandomize(false));
  $('#btnTheatre').addEventListener('click', enterTheatre);
  $('#btnCopyLink').addEventListener('click', async () => {
    pushUrl();
    try { await navigator.clipboard.writeText(location.href); toast('Link copied. It rebuilds this exact look.'); }
    catch (e) { toast('Copy blocked; the link is in the address bar.'); }
  });
  $('#btnHelp').addEventListener('click', showHelp);
  $('#overlayMsg').addEventListener('click', hideOverlay);

  /* presets */
  $('#btnSavePreset').addEventListener('click', () => {
    const nm = ($('#presetName').value || '').trim() || (GEN_BY_ID[S.gen].name + ' ' + S.seed);
    const list = loadPresets();
    list.unshift({ name: nm, data: snapshot() });
    savePresets(list.slice(0, 60)); buildPresetList();
    $('#presetName').value = ''; toast('Saved "' + nm + '"');
  });
  $('#btnExportJson').addEventListener('click', () => {
    const blob = new Blob([JSON.stringify({ presets: loadPresets(), current: snapshot() }, null, 2)], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob); a.download = 'electric-loom-presets.json'; a.click();
  });
  $('#btnImportJson').addEventListener('click', () => $('#fileJson').click());
  $('#fileJson').addEventListener('change', async e => {
    const f = e.target.files[0]; if (!f) return;
    try {
      const o = JSON.parse(await f.text());
      if (o.presets && Array.isArray(o.presets)) {
        savePresets(o.presets.concat(loadPresets()).slice(0, 120));
        buildPresetList(); toast('Imported ' + o.presets.length + ' looks.');
      }
      if (o.current) { applySnapshot(o.current); pushUrl(); scheduleThumbs(); }
      else if (o.gen) { applySnapshot(o); pushUrl(); scheduleThumbs(); }
    } catch (err) { toast('That file did not parse as Electric Loom JSON.'); }
    e.target.value = '';
  });

  /* keyboard */
  window.addEventListener('keydown', e => {
    /* theatre mode captures its own keys, whatever has focus */
    if (TH.active) {
      switch (e.key) {
        case 'Escape': case 't': case 'T': e.preventDefault(); exitTheatre(); break;
        case ' ': e.preventDefault(); toggleTheatrePlay(); break;
        case 'ArrowRight': slideNextManual(); break;
        case 'ArrowLeft': slidePrevManual(); break;
        case 's': case 'S': toggleSlideshow(); break;
      }
      return;
    }
    const tag = (e.target.tagName || '').toLowerCase();
    if (tag === 'input' || tag === 'select' || tag === 'textarea') {
      if (e.key === 'Escape') e.target.blur();
      return;
    }
    const N = totalFrames();
    switch (e.key) {
      case 't': case 'T': if (!busy) enterTheatre(); break;
      case ' ': e.preventDefault(); togglePlay(); break;
      case 'r': case 'R': if (!busy) doRandomize(true); break;
      case 'p': case 'P': if (!busy) doRandomize(false); break;
      case 's': case 'S': toggleSeam(); break;
      case 'm': case 'M':
        S.modMaster.on = !S.modMaster.on;
        $('#modMaster').checked = S.modMaster.on;
        refreshModButtons(); updateModCount(); markDirty();
        toast('Modulation ' + (S.modMaster.on ? 'on' : 'off'));
        break;
      case 'v': case 'V': if (!busy) verifyLoop(); break;
      case 'g': case 'G': if (!busy) exportGif(); break;
      case 'w': case 'W': if (!busy) exportWebm(); break;
      case 'Escape': hideOverlay(); break;
      case 'ArrowRight': playing = false; tNow = (Math.round(tNow * N) + 1) % N / N; markDirty(); syncPlayBtn(); break;
      case 'ArrowLeft': playing = false; tNow = ((Math.round(tNow * N) - 1 + N) % N) / N; markDirty(); syncPlayBtn(); break;
    }
  });

  window.addEventListener('resize', layout);
}
function togglePlay() { playing = !playing; if (playing) seamMode = 0; syncPlayBtn(); }
function syncPlayBtn() {
  $('#btnPlay').innerHTML = playing ? '&#10074;&#10074;' : '&#9654;';
  $('#btnPlay').classList.toggle('on', playing);
  $('#btnSeam').classList.toggle('on', !!seamMode);
}
function toggleSeam() {
  seamMode = seamMode ? 0 : 1;
  if (seamMode) { playing = false; seamTick = 0; }
  else { $('#seamBadge').className = ''; }
  syncPlayBtn(); markDirty();
}

      'Try Chrome, Edge or Firefox, and check that hardware acceleration is on in the browser settings.</p></div></div>';
    return;
  }
  wireSections();
  buildGallery();
  buildSizePresets();
  wire();
  buildStops();
  buildPalPresets();
  uploadPalette();
  drawPalBar();
  buildParams();
  buildModList();
  buildPresetList();
  $('#genTag').textContent = GEN_BY_ID[S.gen].name;

  if (!readUrl()) {
    // a good-looking first impression rather than a blank default
    S.pal.stops = PAL_PRESETS[0].s.map((c, i) => ({ p: i / PAL_PRESETS[0].s.length, c }));
    S.pal.preset = 0;
    S.frame.zoom = 1.15;
    refreshAllUI();
  }
  /* the rail is only shown if an operator actually injected a unit */
  const adRail = $('#adRail');
  if (adRail && adRail.children.length) { HAS_SPONSOR = true; $('#adSec').style.display = ''; }
  else if (adRail) { const s = $('#adSec'); if (s) s.remove(); }

  /* phones get a cheaper preview by default; the selector is hidden there */
  if (isNarrow()) { const pq = $('#previewQ'); if (pq) pq.value = '0.5'; }
  readTipConfig();

  probeCodecs();
  syncPlayBtn();
  layout();
  updateEstimate();
  setTimeout(renderThumbs, 60);
  requestAnimationFrame(tick);

  // tell the user which encoding path they got, once, quietly
  setTimeout(() => {
    const w = getWorkers();
    if (!w) {
      $('#progTxt').textContent = '';
      const note = el('div', 'warnbox');
      note.innerHTML = 'GIF encoding is running on the main thread because background workers are blocked ' +
        'on file:// pages. It still works, just slower. Run <b>serve.cmd</b> in this folder and open ' +
        '<b>http://localhost:8777</b> to get all cores.';
      $('#outList').parentNode.insertBefore(note, $('#outList'));
    }
  }, 400);
})();
